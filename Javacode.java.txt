import java.util.*;
import java.time.*;
import java.time.format.DateTimeFormatter;

// 1. Data Model
class AirQuality {
    double pm25, pm10, co2, o3, temperature, humidity;
    String location;
    LocalDateTime timestamp;

    public AirQuality(double pm25, double pm10, double co2, double o3,
                      double temperature, double humidity, String location) {
        this.pm25 = pm25;
        this.pm10 = pm10;
        this.co2 = co2;
        this.o3 = o3;
        this.temperature = temperature;
        this.humidity = humidity;
        this.location = location;
        this.timestamp = LocalDateTime.now();
    }

    @Override
    public String toString() {
        DateTimeFormatter formatter = DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm:ss");
        return String.format(
            "--- [%s] ---\nLocation: %s\nPM2.5: %.2f | PM10: %.2f\nCO2: %.2f | O3: %.2f\nTemp: %.1f°C | Hum: %.1f%%\n",
            timestamp.format(formatter), location, pm25, pm10, co2, o3, temperature, humidity
        );
    }
}

// 2. Logic Controller
class AQICalculator {
    public static String getAQI(double pm25) {
        if (pm25 <= 12) return "Good (Green)";
        if (pm25 <= 35) return "Moderate (Yellow)";
        if (pm25 <= 55) return "Unhealthy for Sensitive Groups (Orange)";
        if (pm25 <= 150) return "Unhealthy (Red)";
        if (pm25 <= 250) return "Very Unhealthy (Purple)";
        return "Hazardous (Maroon)";
    }
}

// 3. Main Application Class
public class Main { // Renamed to Main for compatibility with most online compilers

    static List<AirQuality> records = new ArrayList<>();
    static Scanner sc = new Scanner(System.in);

    public static void main(String[] args) {
        while (true) {
            try {
                System.out.println("\n=== AIR QUALITY MONITORING SYSTEM ===");
                System.out.println("1. Add Sensor Data");
                System.out.println("2. View History");
                System.out.println("3. Check Latest AQI Status");
                System.out.println("4. Show Average PM2.5");
                System.out.println("5. Exit");
                System.out.print("Select an option: ");

                String input = sc.nextLine(); // Read as string to avoid scanner bugs
                int choice = Integer.parseInt(input);

                switch (choice) {
                    case 1 -> addData();
                    case 2 -> viewRecords();
                    case 3 -> showAQI();
                    case 4 -> averagePM25();
                    case 5 -> {
                        System.out.println("System shutting down. Stay safe!");
                        return;
                    }
                    default -> System.out.println("⚠️ Invalid choice. Please pick 1-5.");
                }
            } catch (NumberFormatException e) {
                System.out.println("⚠️ Error: Please enter a valid number.");
            }
        }
    }

    static void addData() {
        try {
            System.out.print("Enter Location Name: ");
            String location = sc.nextLine();

            System.out.print("Enter PM2.5 (µg/m³): ");
            double pm25 = Double.parseDouble(sc.nextLine());

            System.out.print("Enter PM10: ");
            double pm10 = Double.parseDouble(sc.nextLine());

            System.out.print("Enter CO2 (ppm): ");
            double co2 = Double.parseDouble(sc.nextLine());

            System.out.print("Enter O3: ");
            double o3 = Double.parseDouble(sc.nextLine());

            System.out.print("Enter Temperature (°C): ");
            double temp = Double.parseDouble(sc.nextLine());

            System.out.print("Enter Humidity (%): ");
            double hum = Double.parseDouble(sc.nextLine());

            records.add(new AirQuality(pm25, pm10, co2, o3, temp, hum, location));
            System.out.println("✅ Data recorded successfully!");
        } catch (NumberFormatException e) {
            System.out.println("⚠️ Input Error: Please use numbers for pollutants and weather.");
        }
    }

    static void viewRecords() {
        if (records.isEmpty()) {
            System.out.println("No records found.");
            return;
        }
        System.out.println("\n--- ALL RECORDED DATA ---");
        records.forEach(System.out::println);
    }

    static void showAQI() {
        if (records.isEmpty()) {
            System.out.println("No data available to calculate AQI.");
            return;
        }

        AirQuality latest = records.get(records.size() - 1);
        String category = AQICalculator.getAQI(latest.pm25);

        System.out.println("\n--- LATEST AQI REPORT ---");
        System.out.println("Location: " + latest.location);
        System.out.println("PM2.5 Level: " + latest.pm25);
        System.out.println("Status: " + category);
    }

    static void averagePM25() {
        if (records.isEmpty()) {
            System.out.println("No records found.");
            return;
        }

        double avg = records.stream()
                            .mapToDouble(aq -> aq.pm25)
                            .average()
                            .orElse(0.0);

        System.out.printf("Average PM2.5 across all %d records: %.2f µg/m³\n", records.size(), avg);
    }
}
